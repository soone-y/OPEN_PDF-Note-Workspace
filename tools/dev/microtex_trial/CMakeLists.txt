cmake_minimum_required(VERSION 3.16)
project(microtex_trial LANGUAGES CXX)

# Development-only, offline build. Supply already acquired source directories.
set(MICROTEX_SOURCE_DIR "" CACHE PATH "Fixed MicroTeX source directory")
set(TINYXML2_SOURCE_DIR "" CACHE PATH "Fixed tinyxml2 source directory")
option(MICROTEX_OPENMATH "Use the openmath backend with XITS path resources" OFF)
if(NOT WIN32)
    message(FATAL_ERROR "Windows is required")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(MICROTEX_OPENMATH)
    if(NOT EXISTS "${MICROTEX_SOURCE_DIR}/lib/microtex.cpp")
        message(FATAL_ERROR "A local openmath source directory is required")
    endif()
    set(BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(HAVE_LOG OFF CACHE BOOL "" FORCE)
    set(HAVE_AUTO_FONT_FIND OFF CACHE BOOL "" FORCE)
    set(GLYPH_RENDER_TYPE 1 CACHE STRING "" FORCE)
    add_subdirectory("${MICROTEX_SOURCE_DIR}" upstream)
    set(GDI_SOURCE "${MICROTEX_SOURCE_DIR}/platform/gdi_win/graphic_win32.cpp")
else()
    if(NOT EXISTS "${MICROTEX_SOURCE_DIR}/src/latex.cpp"
                 OR NOT EXISTS "${TINYXML2_SOURCE_DIR}/tinyxml2.cpp")
        message(FATAL_ERROR "Windows and both local source directories are required")
    endif()

    file(GLOB_RECURSE MICROTEX_CORE CONFIGURE_DEPENDS "${MICROTEX_SOURCE_DIR}/src/*.cpp")
    list(FILTER MICROTEX_CORE EXCLUDE REGEX "/(platform|samples)/")
    set(GDI_SOURCE "${MICROTEX_SOURCE_DIR}/src/platform/gdi_win/graphic_win32.cpp")
endif()
if(MINGW)
    # comdef.h introduces a global Font that conflicts with the engine Font class in MinGW.
    # Generate a build-local copy, preserving the acquired upstream source.
    file(READ "${GDI_SOURCE}" GDI_CONTENT)
    string(FIND "${GDI_CONTENT}" "#include <comdef.h>" COMDEF_POSITION)
    if(COMDEF_POSITION EQUAL -1)
        message(FATAL_ERROR "The expected upstream comdef.h include has changed")
    endif()
    string(REPLACE "#include <comdef.h>" "// Unused COM header omitted for the MinGW trial."
        GDI_CONTENT "${GDI_CONTENT}")
    set(GDI_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/graphic_win32_trial.cpp")
    file(WRITE "${GDI_SOURCE}" "#undef TRANSPARENT\n${GDI_CONTENT}")
    set_source_files_properties("${GDI_SOURCE}" PROPERTIES COMPILE_OPTIONS "-include;windows.h")
endif()
if(MICROTEX_OPENMATH)
    add_library(microtex_trial_engine STATIC "${GDI_SOURCE}")
    target_link_libraries(microtex_trial_engine PUBLIC microtex)
    target_include_directories(microtex_trial_engine PUBLIC "${MICROTEX_SOURCE_DIR}/platform/gdi_win")
    target_compile_definitions(microtex_trial_engine PUBLIC MICROTEX_TRIAL_OPENMATH)
else()
    add_library(microtex_trial_engine STATIC ${MICROTEX_CORE}
        "${GDI_SOURCE}"
        "${TINYXML2_SOURCE_DIR}/tinyxml2.cpp")
    target_include_directories(microtex_trial_engine PUBLIC
        "${MICROTEX_SOURCE_DIR}/src" "${TINYXML2_SOURCE_DIR}")
endif()
target_compile_features(microtex_trial_engine PUBLIC cxx_std_17)
target_compile_definitions(microtex_trial_engine PUBLIC BUILD_WIN32 NOMINMAX NDEBUG _HAS_STD_BYTE=0)

add_executable(microtex_trial probe.cpp "${REPO_ROOT}/src/math/math_render.cpp")
target_include_directories(microtex_trial PRIVATE "${REPO_ROOT}/src")
target_link_libraries(microtex_trial PRIVATE microtex_trial_engine gdiplus gdi32 user32)
if(MINGW)
    # Keep the probe independent of unrelated MinGW DLLs earlier on PATH.
    target_link_options(microtex_trial PRIVATE -municode -static)
endif()
if(MSVC)
    target_compile_options(microtex_trial_engine PRIVATE /utf-8)
    target_compile_options(microtex_trial PRIVATE /utf-8)
endif()
